cd zkp_proof

# Compile circuit
circom circuit/voting.circom --r1cs --wasm --sym --c

# Computing witness
node voting_js/generate_witness.js voting_js/voting.wasm ../zkp_interaction/input.json witness.wtns

# Generate Proof
## Ceremony
snarkjs powersoftau new bn128 17 pot17_0000.ptau -v
snarkjs powersoftau contribute pot17_0000.ptau pot17_0001.ptau --name="First contribution" -v
snarkjs powersoftau prepare phase2 pot17_0001.ptau pot17_final.ptau -v
snarkjs groth16 setup voting.r1cs pot17_final.ptau voting_0000.zkey
snarkjs zkey contribute voting_0000.zkey voting_0001.zkey --name="1st Contributor Name" -v
snarkjs zkey export verificationkey voting_0001.zkey verification_key.json

## Generate a proof
snarkjs groth16 prove voting_0001.zkey witness.wtns proof.json public.json

#Verify a proof
snarkjs groth16 verify verification_key.json public.json proof.json

#verifier.sol
snarkjs zkey export solidityverifier voting_0001.zkey verifier.sol
